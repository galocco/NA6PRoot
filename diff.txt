+void NA6PTrackerCA::saveQAHistograms(const std::string& filename)
+{
+  TFile* outFile = TFile::Open(filename.c_str(), "RECREATE");
+  if (!outFile || outFile->IsZombie()) {
+    LOGP(error, "Could not create output file: {}", filename);
+    return;
+  }
+
+  TDirectory* dirTrue = outFile->mkdir("True");
+  TDirectory* dirFake = outFile->mkdir("Fake");
+  TDirectory* dirCommon = outFile->mkdir("Common");
+
+  auto writeCanvasWithCut = [&](TH1F* hist, float cutValue, bool cutIsMin) {
+    if (!hist) {
+      return;
+    }
+    TCanvas canvas(Form("c_%s", hist->GetName()), hist->GetTitle(), 900, 700);
+    hist->SetLineWidth(2);
+    hist->Draw("hist");
+    const double ymax = std::max(1.0, hist->GetMaximum() * 1.1);
+    TLine cutLine(cutValue, 0., cutValue, ymax);
+    cutLine.SetLineColor(cutIsMin ? 4 : 2);
+    cutLine.SetLineWidth(2);
+    cutLine.SetLineStyle(2);
+    cutLine.Draw("same");
+    canvas.Write();
+  };
+
+  for(int it = 0; it < mNIterationsCA; ++it) {
+    for (int truth = 0; truth < kQATruthCats; ++truth) {
+      TDirectory* dirTopTruth = (truth == kQATrue) ? dirTrue : dirFake;
+      if (!dirTopTruth) {
+        continue;
+      }
+      TDirectory* dirIter = dirTopTruth->mkdir(Form("Iter%d", it));
+      if (!dirIter) {
+        continue;
+      }
+      dirIter->cd();
+
+      for (int iLayer = 0; iLayer < mNLayers - 1 && iLayer < kMaxQALayers; ++iLayer) {
+        if (mHistDeltaThetaTracklets[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistDeltaThetaTracklets[truth][it][iLayer], mMaxDeltaThetaTrackletsCA[it], false);
+        }
+        if (mHistDeltaPhiTracklets[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistDeltaPhiTracklets[truth][it][iLayer], mMaxDeltaPhiTrackletsCA[it], false);
+        }
+      }
+      for (int iLayer = 0; iLayer < mNLayers - 2 && iLayer < kMaxQALayers; ++iLayer) {
+        if (mHistDeltaTanLCells[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistDeltaTanLCells[truth][it][iLayer], mMaxDeltaTanLCellsCA[it], false);
+        }
+        if (mHistDeltaPhiCells[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistDeltaPhiCells[truth][it][iLayer], mMaxDeltaPhiCellsCA[it], false);
+        }
+        if (mHistDeltaPxPzCells[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistDeltaPxPzCells[truth][it][iLayer], mMaxDeltaPxPzCellsCA[it], false);
+        }
+        if (mHistDeltaPyPzCells[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistDeltaPyPzCells[truth][it][iLayer], mMaxDeltaPyPzCellsCA[it], false);
+        }
+        if (mHistChi2TrClCells[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistChi2TrClCells[truth][it][iLayer], mMaxChi2TrClCellsCA[it], false);
+        }
+        if (mHistChi2ndfCells[truth][it][iLayer]) {
+          writeCanvasWithCut(mHistChi2ndfCells[truth][it][iLayer], mMaxChi2ndfCellsCA[it], false);
+        }
+      }
+    }
+
+    if (dirCommon) {
+      TDirectory* dirIterCommon = dirCommon->mkdir(Form("Iter%d", it));

+        dirIterCommon->cd();
+      }
